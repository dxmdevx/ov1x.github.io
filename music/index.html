Setup Sheet Editing

Deploy this Google Apps Script as a Web App, then paste the Web App URL below. The site uses it to add, update, and delete rows after a valid code is entered.

  1. Open your Google Sheet: Open Spreadsheet.
  2. In the top menu, select Extensions > Apps Script.
  3. Delete any code in the editor and paste the script below.
  4. Deploy as a Web app, set access to Anyone, then copy the Web App URL.
function doPost(e) {
  var body = JSON.parse((e.postData && e.postData.contents) || '{}');
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var action = String(body.action || '').toLowerCase();
  var song = body.song || {};
  var rowNumber = Number(body.rowNumber || 0);

  if (!body.code) return reply({ ok: false, error: 'Missing access code' });
  if (!action) return reply({ ok: false, error: 'Missing action' });

  var id = String(song.id || '').trim();
  var name = String(song.name || '').trim();
  var category = String(song.category || '').trim();
  var now = new Date();

  if (action === 'add') {
    if (!id || !name) return reply({ ok: false, error: 'ID and Name are required' });
    sheet.appendRow([id, name, category, now]);
    return reply({ ok: true });
  }

  if (action === 'update') {
    if (rowNumber < 2) return reply({ ok: false, error: 'Invalid row number' });
    if (!id || !name) return reply({ ok: false, error: 'ID and Name are required' });
    sheet.getRange(rowNumber, 1, 1, 4).setValues([[id, name, category, now]]);
    return reply({ ok: true });
  }

  if (action === 'delete') {
    if (rowNumber < 2) return reply({ ok: false, error: 'Invalid row number' });
    sheet.deleteRow(rowNumber);
    return reply({ ok: true });
  }

  return reply({ ok: false, error: 'Unknown action' });
}

function reply(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
